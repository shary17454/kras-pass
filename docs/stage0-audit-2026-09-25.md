```sh

godot --headless --path . --log-file /tmp/kras-stage0-games.log tools/party_content_audit.tscn -- --test-data-dir=/tmp/kras-stage0-games --recheck=res://build/party/race-balance/report.json --recheck=res://build/party/gem-balance/report.json
```
